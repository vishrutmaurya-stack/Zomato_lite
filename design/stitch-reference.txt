<script>
  function switchState(state) {
    const states = ['loading', 'success', 'error'];
    states.forEach(s => {
      const el = document.getElementById(`state-${s}`);
      const tab = document.getElementById(`tab-${s}`);
      if (s === state) {
        el.classList.remove('hidden');
        el.classList.add('flex');
        tab.className = "flex-1 py-2 px-space-xs rounded-lg font-label-md text-label-md transition-all text-on-surface bg-surface-container-lowest shadow-sm flex items-center justify-center gap-1";
      } else {
        el.classList.add('hidden');
        el.classList.remove('flex');
        tab.className = "flex-1 py-2 px-space-xs rounded-lg font-label-md text-label-md transition-all text-on-surface-variant hover:text-on-surface flex items-center justify-center gap-1";
      }
    });
  }

  function handleRetry() {
    const btn = document.getElementById('btn-retry');
    const originalContent = btn.innerHTML;
    btn.innerHTML = '<span class="material-symbols-outlined text-[20px] animate-spin">progress_activity</span><span>Reconnecting...</span>';
    btn.disabled = true;
    setTimeout(() => {
      btn.innerHTML = originalContent;
      btn.disabled = false;
      switchState('success');
    }, 1500);
  }
